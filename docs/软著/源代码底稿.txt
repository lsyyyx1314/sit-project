     1	// MPV-2：面向 SDN 多路径的可验证定长验证头。
     2	//
     3	// 与 Python 参考实现（mpv/core/scheme.py）逐位一致，交叉校验见
     4	// cpp/tests/golden_vectors.txt。
     5	//
     6	// 核心思想：把随机性放在**线性项**而不是指数上，控制器直接代入即可，
     7	// 不存在"消去随机数"的问题，也就不会出现 g^w ≡ 1 的恒真退化。

    10	//     c_i  = (a_i + b_i·h) mod p           单跳贡献
    11	//     V    = Σ c_i · ρ^(i-1)  mod p        逐跳累积

   126	struct FlowAgg {
   127	  uint64_t n = 0;
   128	  uint64_t sum_h = 0;
   129	  uint64_t sum_v = 0;
   130	  void add(uint64_t nonce, uint64_t v) {
   131	    ++n;
   132	    sum_h = addmod(sum_h, h_of(nonce));

   134	  }
   135	  void reset() { n = sum_h = sum_v = 0; }
   136	};

   146	  bool see(uint64_t nonce);

    28	inline uint64_t rotr64(uint64_t x, int n) {
    29	  return (x >> n) | (x << (64 - n));
    30	}
    31	
    32	inline uint64_t load64(const uint8_t* p) {
    33	  uint64_t v = 0;
    34	  for (int i = 7; i >= 0; --i) v = (v << 8) | p[i];
    35	  return v;
    36	}
    37	
    38	inline void store64(uint8_t* p, uint64_t v) {
    39	  for (int i = 0; i < 8; ++i) {
    40	    p[i] = static_cast<uint8_t>(v & 0xff);
    41	    v >>= 8;
    42	  }
    43	}

    45	inline void G(uint64_t* v, int a, int b, int c, int d, uint64_t x, uint64_t y) {
    46	  v[a] = v[a] + v[b] + x;
    47	  v[d] = rotr64(v[d] ^ v[a], 32);
    48	  v[c] = v[c] + v[d];
    49	  v[b] = rotr64(v[b] ^ v[c], 24);
    50	  v[a] = v[a] + v[b] + y;
    51	  v[d] = rotr64(v[d] ^ v[a], 16);
    52	  v[c] = v[c] + v[d];
    53	  v[b] = rotr64(v[b] ^ v[c], 63);
    54	}

    56	void compress(uint64_t* h, const uint8_t block[128], uint64_t t, bool last) {
    57	  uint64_t v[16];
    58	  uint64_t m[16];
    59	  for (int i = 0; i < 8; ++i) v[i] = h[i];
    60	  for (int i = 0; i < 8; ++i) v[i + 8] = IV[i];
    61	  v[12] ^= t;
    62	  v[13] ^= 0;
    63	  if (last) v[14] = ~v[14];
    64	  for (int i = 0; i < 16; ++i) m[i] = load64(block + i * 8);
    65	
    66	  for (int r = 0; r < 12; ++r) {
    67	    const uint8_t* s = SIGMA[r];
    68	    G(v, 0, 4, 8, 12, m[s[0]], m[s[1]]);
    69	    G(v, 1, 5, 9, 13, m[s[2]], m[s[3]]);
    70	    G(v, 2, 6, 10, 14, m[s[4]], m[s[5]]);
    71	    G(v, 3, 7, 11, 15, m[s[6]], m[s[7]]);
    72	    G(v, 0, 5, 10, 15, m[s[8]], m[s[9]]);
    73	    G(v, 1, 6, 11, 12, m[s[10]], m[s[11]]);
    74	    G(v, 2, 7, 8, 13, m[s[12]], m[s[13]]);
    75	    G(v, 3, 4, 9, 14, m[s[14]], m[s[15]]);
    76	  }
    77	  for (int i = 0; i < 8; ++i) h[i] ^= v[i] ^ v[i + 8];
    78	}

    82	void blake2b(uint8_t* out, size_t outlen, const void* in, size_t inlen,
    83	             const void* key, size_t keylen) {
    84	  uint64_t h[8];
    85	  std::memcpy(h, IV, sizeof(h));
    86	  h[0] ^= 0x01010000ULL ^ (static_cast<uint64_t>(keylen) << 8) ^
    87	          static_cast<uint64_t>(outlen);
    88	
    89	  uint8_t block[128];
    90	  uint64_t t = 0;
    91	  const uint8_t* msg = static_cast<const uint8_t*>(in);
    92	
    93	  if (keylen > 0) {
    94	    std::memset(block, 0, sizeof(block));
    95	    std::memcpy(block, key, keylen);
    96	    t += 128;
    97	    compress(h, block, t, inlen == 0);
    98	    if (inlen == 0) {
    99	      // 密钥块即最后一块
   100	      for (size_t i = 0; i < outlen; ++i)
   101	        out[i] = static_cast<uint8_t>((h[i / 8] >> (8 * (i % 8))) & 0xff);
   102	      return;
   103	    }
   104	  }
   105	
   106	  size_t i = 0;
   107	  while (inlen - i > 128) {
   108	    t += 128;
   109	    compress(h, msg + i, t, false);
   110	    i += 128;
   111	  }
   112	  size_t rem = inlen - i;
   113	  std::memset(block, 0, sizeof(block));
   114	  std::memcpy(block, msg + i, rem);
   115	  t += rem;
   116	  compress(h, block, t, true);
   117	
   118	  for (size_t k = 0; k < outlen; ++k)
   119	    out[k] = static_cast<uint8_t>((h[k / 8] >> (8 * (k % 8))) & 0xff);
   120	}
   121	
   122	uint64_t blake2b_u64(const void* in, size_t inlen) {
   123	  uint8_t buf[8];
   124	  blake2b(buf, 8, in, inlen, nullptr, 0);
   125	  uint64_t v = 0;
   126	  for (int i = 0; i < 8; ++i) v = (v << 8) | buf[i];  // 大端
   127	  return v;
   128	}

    21	// keys.py::_derive —— 无密钥 blake2b，输入 seed(32B) || label
    22	uint64_t derive(uint64_t seed, const std::string& label) {
    23	  uint8_t buf[32 + 128];
    24	  put_u64_be32(buf, seed);
    25	  size_t n = label.size() > 128 ? 128 : label.size();
    26	  std::memcpy(buf + 32, label.data(), n);
    27	  return blake2b_u64(buf, 32 + n);

    31	uint64_t prf(uint64_t key, const char* label, size_t len) {
    32	  uint8_t k[32];

    34	  uint8_t out[8];

    36	  uint64_t v = 0;
    37	  for (int i = 0; i < 8; ++i) v = (v << 8) | out[i];

    44	uint64_t mulmod(uint64_t a, uint64_t b) {
    45	  return static_cast<uint64_t>((static_cast<unsigned __int128>(a) * b) % P);

    48	uint64_t addmod(uint64_t a, uint64_t b) {
    49	  // 直接用 128 位中间量，避免 a+b 溢出 2^64 时的手工修正
    50	  return static_cast<uint64_t>(
    51	      (static_cast<unsigned __int128>(a) + static_cast<unsigned __int128>(b)) % P);
    52	}

    54	uint64_t powmod(uint64_t base, uint64_t exp) {
    55	  uint64_t r = 1 % P;
    56	  base %= P;
    57	  while (exp) {
    58	    if (exp & 1) r = mulmod(r, base);
    59	    base = mulmod(base, base);
    60	    exp >>= 1;
    61	  }
    62	  return r;
    63	}

    66	KeyRing KeyRing::generate(uint32_t n_switches, uint32_t ports_per_switch,
    67	                          uint64_t seed) {
    68	  KeyRing kr;

    75	  }
    76	  return kr;

   101	  put_u56_be(out, h.nonce);

   108	  Header h{};

   110	  for (int i = 0; i < 8; ++i) h.acc = (h.acc << 8) | in[7 + i];

   129	Header advance(const KeyRing& kr, const Hop& h, const Header& hdr, uint32_t pos) {
   130	  uint64_t c = contribution(kr, h, h_of(hdr.nonce));
   131	  Header out = hdr;
   132	  out.acc = addmod(hdr.acc, mulmod(c, powmod(RHO, pos - 1)));

   136	Header forward(const KeyRing& kr, const Path& path, uint8_t tag, uint64_t nonce) {
   137	  Header hdr = ingress(kr, path.front(), tag, nonce);
   138	  for (size_t i = 1; i < path.size(); ++i)
   139	    hdr = advance(kr, path[i], hdr, static_cast<uint32_t>(i + 1));
   140	  return hdr;

   144	Expectation expectation(const KeyRing& kr, const Path& path) {
   145	  uint64_t A = 0, B = 0, w = 1;
   146	  for (const Hop& h : path) {
   147	    HopKey k = kr.hop_key(h);
   148	    A = addmod(A, mulmod(k.a, w));
   149	    B = addmod(B, mulmod(k.b, w));
   150	    w = mulmod(w, RHO);
   151	  }
   152	  return Expectation{A, B};
   153	}

   156	                                   const std::vector<Path>& paths)
   157	    : paths_(paths) {
   158	  table_.reserve(paths.size());

   162	Verdict ExpectationTable::verify_tag(const Header& hdr) const {
   163	  Verdict v;
   164	  if (hdr.tag >= table_.size()) return v;
   165	  uint64_t hh = h_of(hdr.nonce);
   166	  v.checks = 1;
   167	  if (table_[hdr.tag].expect(hh) == hdr.acc) {
   168	    v.ok = true;
   169	    v.matched = hdr.tag;
   170	  }
   171	  return v;
   172	}

   174	Verdict ExpectationTable::verify_scan(const Header& hdr) const {
   175	  Verdict v;
   176	  uint64_t hh = h_of(hdr.nonce);
   177	  for (size_t i = 0; i < table_.size(); ++i) {
   178	    v.checks = static_cast<int>(i + 1);
   179	    if (table_[i].expect(hh) == hdr.acc) {
   180	      v.ok = true;
   181	      v.matched = static_cast<int>(i);
   182	      return v;
   183	    }
   184	  }
   185	  return v;
   186	}

   188	// ---------------------------------------------------------------- CFMV
   189	// 判定式：N·A_P + B_P·Σh ≡ ΣV (mod p)
   190	// 来源：Σ_i (A + B·h_i) = N·A + B·Σh，线性性把 N 次逐包判定压成 1 次。
   191	Verdict ExpectationTable::verify_flow(uint8_t tag, uint64_t n, uint64_t sum_h,
   192	                                      uint64_t sum_v) const {
   193	  Verdict v;
   194	  if (tag >= table_.size() || n == 0) return v;
   195	  v.checks = 1;
   196	  const Expectation& e = table_[tag];
   197	  uint64_t lhs = addmod(mulmod(n % P, e.A), mulmod(e.B, sum_h));
   198	  if (lhs == sum_v) {
   199	    v.ok = true;
   200	    v.matched = tag;
   201	  }
   202	  return v;
   203	}
   204	
   205	Verdict ExpectationTable::verify_flow_scan(uint64_t n, uint64_t sum_h,
   206	                                           uint64_t sum_v) const {
   207	  Verdict v;
   208	  if (n == 0) return v;
   209	  for (size_t i = 0; i < table_.size(); ++i) {
   210	    v.checks = static_cast<int>(i + 1);
   211	    const Expectation& e = table_[i];
   212	    uint64_t lhs = addmod(mulmod(n % P, e.A), mulmod(e.B, sum_h));
   213	    if (lhs == sum_v) {
   214	      v.ok = true;
   215	      v.matched = static_cast<int>(i);
   216	      return v;
   217	    }
   218	  }
   219	  return v;
   220	}

   223	// 环形缓冲线性扫描：容量 4096、包速率 ≤ 10^5/s 时，平均扫描 ~2000 次比较，
   224	// 远小于一次 blake2b；不需要哈希表。
   225	bool NonceWindow::see(uint64_t nonce) {
   226	  for (size_t i = 0; i < ring_.size(); ++i) {
   227	    size_t idx = (head_ + i) % ring_.size();
   228	    if (ring_[idx] == nonce && ring_[idx] != 0) return false;
   229	  }
   230	  ring_[head_] = nonce;
   231	  head_ = (head_ + 1) % ring_.size();
   232	  return true;
   233	}

    46	Event Controller::feed_packet(std::istringstream& ss) {
    47	  Event ev;
    48	  unsigned sid, tag;
    49	  unsigned long long nonce, acc;
    50	  char mac[32];
    51	  if (std::sscanf((ss.str()).c_str(), "%u %u %llu %llu %31s", &sid, &tag,
    52	                  &nonce, &acc, mac) != 5)
    53	    return ev;
    54	  ev.kind = "packet";
    55	  ev.mac = mac;
    56	  ev.nonce = nonce;
    57	  ev.acc = acc;
    58	  ev.tag = tag;
    59	
    60	  int d = dir_of(mac);
    61	  if (d < 0) return ev;
    62	  DirStat& st = stats_[(size_t)d];
    63	  ev.claimed = (int)tag;
    64	
    65	  // 重放窗口：同方向最近见过的 nonce 直接判重放，不进入正常判定
    66	  if (!windows_[(size_t)d].see(nonce)) {
    67	    ev.replay = true;
    68	    ++st.replays;
    69	    // 重放是已验证包的复制品，不需要定位；重复的 acc 也不计入通过/失败
    70	    return ev;
    71	  }
    72	
    73	  ++st.packets;
    74	  const Header h{nonce, acc, (uint8_t)tag};
    75	
    76	  double t0 = now_us();
    77	  Verdict v = tables_[(size_t)d].verify_tag(h);
    78	  double t1 = now_us();
    79	  ev.verify_us = t1 - t0;
    80	  ++st.cpmv_checks;
    81	  // BPMV 对照口径：同样的包用 scan 模式再判定一次，记录真实检查次数
    82	  Verdict sc = tables_[(size_t)d].verify_scan(h);
    83	  st.bpmv_checks += (uint64_t)sc.checks;
    84	
    85	  if (v.ok) {
    86	    ++st.pass;
    87	    ev.ok = true;
    88	    ev.matched = v.matched;
    89	    return ev;
    90	  }
    91	  ++st.fail;
    92	  ev.matched = sc.matched;
    93	  ev.suspects = localize_all((size_t)d, (uint8_t)tag, nonce, acc);
    94	  if (!ev.suspects.empty()) {
    95	    ev.suspect_sid = ev.suspects.front();
    96	    ++st.localized;
    97	  }
    98	  return ev;
    99	}

   152	int Controller::localize(size_t dir, uint8_t tag, uint64_t nonce,
   153	                         uint64_t acc) {
   154	  std::vector<int> s = localize_all(dir, tag, nonce, acc);
   155	  return s.empty() ? -1 : s.front();
   156	}
   157	
   158	// 多故障定位：沿实际取证路由逐跳对照。
   159	//   * 篡改检查：期望值用「上一跳的观测值」做基准（篡改偏移 δ 会随帧传播，
   160	//     用纯净累计值会把下游诚实交换机也误标为嫌疑）；
   161	//   * 位置检查：取证的实际 (sid,in,out) 与声称路径第 i 跳不一致 → 改道/改端口
   162	//     嫌疑 = 声称路径的第 i 跳交换机；
   163	//   * 取证覆盖全程但最终上报仍不符 → 出口上报篡改。
   164	std::vector<int> Controller::localize_all(size_t dir, uint8_t tag,
   165	                                          uint64_t nonce, uint64_t acc) {
   166	  std::vector<int> out;
   167	  if (dir >= dirs_.size()) return out;
   168	  const Path& claimed = dirs_[dir].paths[tag];
   169	  const std::string key = dirs_[dir].mac + "|" + std::to_string(nonce);
   170	  auto it = traces_.find(key);
   171	
   172	  uint64_t hh = h_of(nonce);
   173	  if (it == traces_.end() || it->second.empty()) {
   174	    // 无取证：候选对比，声称路径与实际命中路径的首个分歧交换机
   175	    ExpectationTable tb(kr_, dirs_[dir].paths);
   176	    Verdict sc = tb.verify_scan(Header{nonce, acc, tag});
   177	    if (sc.matched < 0) return out;
   178	    const Path& actual = dirs_[dir].paths[(size_t)sc.matched];
   179	    size_t common = 0;
   180	    while (common < claimed.size() && common < actual.size() &&
   181	           claimed[common].sid == actual[common].sid)
   182	      ++common;
   183	    out.push_back((int)claimed[common > 0 ? common - 1 : 0].sid);
   184	    return out;
   185	  }
   186	
   187	  std::vector<HopTrace> recs = it->second;
   188	  std::stable_sort(recs.begin(), recs.end(),
   189	                   [](const HopTrace& a, const HopTrace& b) {
   190	                     return a.pos < b.pos;
   191	                   });
   192	
   193	  uint64_t carry = 0;      // 上一跳之后帧里的实际累加值
   194	  bool on_claimed = true;  // 尚未偏离声称路径
   195	  for (size_t i = 0; i < recs.size(); ++i) {
   196	    const HopTrace& r = recs[i];
   197	    Hop hop{r.sid, r.in_port, r.out_port};
   198	    uint64_t expect = addmod(carry, mulmod(contribution(kr_, hop, hh),
   199	                                           powmod(RHO, (uint64_t)i)));
   200	    if (r.v_after != expect) out.push_back((int)r.sid);  // 本跳篡改嫌疑
   201	    carry = r.v_after;  // 以观测值为准，让后续跳的期望建立在实际上
   202	
   203	    if (on_claimed) {
   204	      if (i < claimed.size() && r.sid == claimed[i].sid &&
   205	          r.in_port == claimed[i].in_port && r.out_port == claimed[i].out_port)
   206	        continue;
   207	      // 与声称路径第 i 跳不一致：改道/改端口发生在该跳
   208	      if (i < claimed.size()) out.push_back((int)claimed[i].sid);
   209	      on_claimed = false;
   210	    }
   211	  }
   212	
   213	  // 取证全程与声称路径吻合、无任何其他嫌疑，但出口上报仍不符：
   214	  // 出口上报本身被篡改。路由已偏离或已有篡改嫌疑时，上报值不等属于正常
   215	  // 推论，不得再把出口交换机标为嫌疑。
   216	  if (out.empty() && on_claimed && recs.size() == claimed.size()) {
   217	    uint64_t expect_total = 0;
   218	    for (size_t j = 0; j < claimed.size(); ++j)
   219	      expect_total =
   220	          addmod(expect_total, mulmod(contribution(kr_, claimed[j], hh),
   221	                                      powmod(RHO, (uint64_t)j)));
   222	    if (acc != expect_total) out.push_back((int)claimed.back().sid);
   223	  }
   224	
   225	  // 去重保序
   226	  std::vector<int> dedup;
   227	  for (int sid : out)
   228	    if (dedup.empty() || dedup.back() != sid) dedup.push_back(sid);
   229	  return dedup;
   230	}

     1	// MPV-2 软件虚拟交换机（用户态数据平面，C++ 实现）。
     2	//
     3	// 为什么不用 OVS：OVS 动作集没有模运算，算不了 c_i = (a_i + b_i·H(nonce)) mod p。
     4	// 为什么不用 BMv2：其 ALU 为 32 位，装不下 64 位模域。因此数据面自己实现。
     5	//
     6	// 帧格式（IPv4 场景）：
     7	//     eth(14) | shim(16) | ip                 shim = nonce(7) V(8) tag(1)
     8	//     ethertype = 0x88B5；出口交换机剥离并恢复 0x0800
     9	//
    10	// 转发表（纯文本，每行一条）：
    11	//     <dst_mac> <tag> <out_port> <pos> <strip> <in_port>
    12	// 以 # 开头为注释。文件 mtime 变化时自动重载。
    13	//
    14	// 上报（出口交换机，每包一行）：
    15	//     <sid> <tag> <nonce> <acc> <dst_mac>
    16	// 流级聚合上报（出口交换机，每凑满 --flow-window 包一行，CFMV）：
    17	//     FLOW <dst_mac> <tag> <N> <sum_h> <sum_v>
    18	//
    19	// 逐跳取证（--trace 开启时，每跳一行，供控制器定位首分歧交换机）：
    20	//     TRACE <dst_mac> <sid> <in> <out> <tag> <nonce> <pos> <V_after>
    21	//
    22	// 恶意交换机模拟（--evil）：
    23	//     bitflip 转发时翻转验证头 acc 的 1 个比特（伪造/篡改）
    24	//     dup     把带验证头的帧原样重发一份（重放攻击）
    25	//
    26	// 用法：
    27	//     vswitch --sid 1 --seed 20260501 --table s1.tbl --report rep.txt
    28	//             --paths 2 --force-tag 0 --port 1:s1p1 --port 2:s1p2

   134	    std::vector<pollfd> pfds;
   135	    for (const Port& p : ports_) pfds.push_back(pollfd{p.fd, POLLIN, 0});
   136	    uint32_t last_check = 0;
   137	    while (true) {

   139	      if (n > 0) {

   150	  void load_table(bool force) {
   151	    struct stat st{};
   152	    if (::stat(table_path_.c_str(), &st) != 0) return;
   153	    if (!force && st.st_mtime == mtime_) return;
   154	    mtime_ = st.st_mtime;
   155	    entries_.clear();
   156	    std::FILE* f = std::fopen(table_path_.c_str(), "r");
   157	    if (!f) return;
   158	    char line[256];
   159	    while (std::fgets(line, sizeof(line), f)) {
   160	      if (line[0] == '#' || line[0] == '\n') continue;
   161	      char mac[32];
   162	      int tag, port, pos, strip, in_port;
   163	      if (std::sscanf(line, "%31s %d %d %d %d %d", mac, &tag, &port, &pos, &strip,
   164	                      &in_port) != 6)
   165	        continue;
   166	      entries_[std::string(mac) + "|" + std::to_string(tag)] =
   167	          Entry{port, pos, strip != 0, in_port};
   168	    }
   169	    std::fclose(f);
   170	  }

   177	  void drain(const Port& p) {
   178	    uint8_t buf[65536];
   179	    while (true) {
   180	      struct sockaddr_ll sll{};
   181	      socklen_t sl = sizeof(sll);
   182	      ssize_t n = ::recvfrom(p.fd, buf, sizeof(buf), MSG_DONTWAIT,
   183	                             reinterpret_cast<sockaddr*>(&sll), &sl);
   184	      if (n < 0) {
   185	        if (errno == EAGAIN || errno == EWOULDBLOCK) break;
   186	        break;
   187	      }
   188	      if (n == 0) break;  // 接口被删除等异常：0 帧不是合法以太帧，避免空转
   189	      if (sll.sll_pkttype == PACKET_OUTGOING) continue;  // 过滤本端口发出的帧
   190	      ++in_;
   191	      handle(buf, static_cast<size_t>(n), p.index);
   192	    }

   195	  void handle(const uint8_t* frame, size_t len, int in_idx) {
   196	    if (len < 14) return;
   197	    uint16_t ether = (uint16_t)((frame[12] << 8) | frame[13]);
   198	    std::string dst = mac_to_str(frame);
   199	    std::string src = mac_to_str(frame + 6);
   200	
   201	    uint8_t tag = 0;
   202	    uint64_t nonce = 0, acc = 0;
   203	    const uint8_t* payload = nullptr;
   204	    size_t payload_len = 0;
   205	    bool shimmed = false;
   206	
   207	    if (ether == kShimEther) {
   208	      if (len < 14 + HEADER_BYTES) return;
   209	      Header h = decode_header(frame + 14);
   210	      nonce = h.nonce;
   211	      acc = h.acc;
   212	      tag = h.tag;
   213	      payload = frame + 14 + HEADER_BYTES;
   214	      payload_len = len - 14 - HEADER_BYTES;
   215	      shimmed = true;
   216	    } else if (ether == kIpEther) {
   217	      payload = frame + 14;
   218	      payload_len = len - 14;
   219	    } else {
   220	      ++drop_;
   221	      return;
   222	    }
   223	
   224	    const Entry* e = nullptr;
   225	    if (!shimmed) {
   226	      if (force_tag_ >= 0) {
   227	        tag = (uint8_t)force_tag_;
   228	      } else {
   229	        uint64_t mix = 1469598103934665603ULL;
   230	        for (int i = 0; i < 6; ++i) mix = (mix ^ frame[6 + i]) * 1099511628211ULL;
   231	        for (int i = 0; i < 6; ++i) mix = (mix ^ frame[i]) * 1099511628211ULL;
   232	        tag = (uint8_t)(mix % (uint64_t)(paths_ > 0 ? paths_ : 1));
   233	      }
   234	    }
   235	    e = lookup(dst, tag);
   236	    if (!e) {
   237	      ++drop_;
   238	      return;
   239	    }
   240	
   241	    Hop hop{sid_, (uint32_t)(e->in_port > 0 ? e->in_port : in_idx),
   242	            (uint32_t)e->port};
   243	    // 入端口以实际到达端口为准（与控制器侧一致时表里的 in_port 应等于它）
   244	    hop.in_port = (uint32_t)in_idx;
   245	
   246	    if (!shimmed) {
   247	      nonce = 0;
   248	      for (int i = 0; i < 7; ++i) nonce = (nonce << 8) | (uint8_t)(rand() & 0xff);
   249	      Header h = ingress(kr_, hop, tag, nonce);
   250	      acc = h.acc;
   251	    } else {
   252	      uint64_t c = contribution(kr_, hop, h_of(nonce));
   253	      acc = addmod(acc, mulmod(c, powmod(RHO, (uint64_t)e->pos - 1)));
   254	    }
   255	
   256	    // 恶意交换机：篡改验证头（在本跳的合法贡献之后叠加比特翻转）
   257	    if (evil_ == Evil::kBitflip && shimmed) acc ^= 1;
   258	
   259	    int out_fd = fd_by_port_[e->port];
   260	    if (out_fd < 0) return;
   261	
   262	    uint8_t out[65536];
   263	    std::memcpy(out, frame, 12);
   264	    uint16_t et = e->strip ? kIpEther : kShimEther;
   265	    out[12] = (uint8_t)(et >> 8);
   266	    out[13] = (uint8_t)(et & 0xff);
   267	    size_t off = 14;
   268	    if (!e->strip) {
   269	      Header h{nonce, acc, tag};
   270	      encode_header(h, out + 14);
   271	      off += HEADER_BYTES;
   272	    }
   273	    std::memcpy(out + off, payload, payload_len);
   274	    struct sockaddr_ll dst_addr{};
   275	    dst_addr.sll_family = AF_PACKET;
   276	    dst_addr.sll_ifindex = ifindex_by_port_[e->port];
   277	    dst_addr.sll_halen = 6;
   278	    std::memcpy(dst_addr.sll_addr, out, 6);
   279	    ssize_t sent = ::sendto(out_fd, out, off + payload_len, 0,
   280	                            reinterpret_cast<sockaddr*>(&dst_addr),
   281	                            sizeof(dst_addr));
   282	    if (sent > 0)
   283	      ++out_;
   284	    else
   285	      std::fprintf(stderr, "[s%u] sendto 失败: %s\n", sid_, std::strerror(errno));
   286	
   287	    // 恶意交换机：重放（把带验证头的帧原样再发一份）
   288	    if (evil_ == Evil::kDup && shimmed && sent > 0)
   289	      ::sendto(out_fd, out, off + payload_len, 0,
   290	               reinterpret_cast<sockaddr*>(&dst_addr), sizeof(dst_addr));
   291	
   292	    // 逐跳取证：记录本跳之后实际写进帧里的累加值
   293	    if (trace_) {
   294	      std::fprintf(trace_, "TRACE %s %u %u %u %u %llu %u %llu\n", dst.c_str(),
   295	                   sid_, hop.in_port, e->port, tag,
   296	                   (unsigned long long)nonce, e->pos,
   297	                   (unsigned long long)acc);
   298	      std::fflush(trace_);
   299	    }
   300	
   301	    if (e->strip) {
   302	      std::fprintf(report_, "%u %u %llu %llu %s\n", sid_, tag,
   303	                   (unsigned long long)nonce, (unsigned long long)acc,
   304	                   dst.c_str());
   305	      std::fflush(report_);
   306	      // CFMV：按 (流, tag) 聚合，凑满一个窗口上报一行
   307	      FlowAgg& a = flows_[{dst, std::to_string(tag)}];
   308	      a.add(nonce, acc);
   309	      if (a.n >= flow_window_) {
   310	        std::fprintf(report_, "FLOW %s %u %llu %llu %llu\n", dst.c_str(), tag,
   311	                     (unsigned long long)a.n, (unsigned long long)a.sum_h,
   312	                     (unsigned long long)a.sum_v);
   313	        a.reset();
   314	      }
   315	    }
   316	    (void)src;
   317	  }

   340	int open_port(const std::string& name) {
   341	  int fd = ::socket(AF_PACKET, SOCK_RAW, htons(ETH_P_ALL));
   342	  if (fd < 0) {
   343	    std::perror("socket");
   344	    return -1;
   345	  }
   346	  struct sockaddr_ll sll{};
   347	  sll.sll_family = AF_PACKET;
   348	  sll.sll_protocol = htons(ETH_P_ALL);
   349	  sll.sll_ifindex = (int)::if_nametoindex(name.c_str());
   350	  if (sll.sll_ifindex == 0) {
   351	    std::fprintf(stderr, "找不到接口 %s\n", name.c_str());
   352	    ::close(fd);
   353	    return -1;
   354	  }
   355	  if (::bind(fd, reinterpret_cast<sockaddr*>(&sll), sizeof(sll)) < 0) {
   356	    std::perror("bind");
   357	    ::close(fd);
   358	    return -1;
   359	  }
   360	  return fd;
   361	}

    27	std::vector<Direction> load_cand(const std::string& file) {
    28	  std::vector<Direction> dirs;
    29	  std::ifstream in(file);
    30	  std::string line;
    31	  while (std::getline(in, line)) {

    44	      std::string tok;

    46	        int sid = 0, ip = 0, op = 0;

    48	        p.push_back(Hop{(uint32_t)sid, (uint32_t)ip, (uint32_t)op});
    49	      }
    50	      dirs.back().paths.push_back(p);
    51	    }
    52	  }
    53	  return dirs;

   101	  }

    29	Path rand_path(Rng& r, uint32_t n_hops) {
    30	  Path p;
    31	  p.push_back(Hop{1, 1, 2});
    32	  std::vector<uint32_t> pool;
    33	  for (uint32_t s = 2; s <= 15; ++s) pool.push_back(s);
    34	  for (size_t i = pool.size(); i > 1; --i) {
    35	    size_t j = r.next() % i;
    36	    std::swap(pool[i - 1], pool[j]);
    37	  }
    38	  for (uint32_t i = 0; i + 1 < n_hops; ++i)
    39	    p.push_back(Hop{pool[i], 1, (uint32_t)(2 + (i % 2))});
    40	  p.push_back(Hop{16, 1, 2});

   160	      ++rows[a].total;
   161	      // 关键：用**合法路径**的表去验证被篡改的包，而不是用篡改后的表
   162	      Header bad = forward(kr, attacked[a], 0, nonce);
   163	      if (!tb.verify_tag(bad).ok) ++rows[a].detected;
   164	    }
   165	  }

   168	  int flip_total = 0, flip_det = 0, forge_total = 0, forge_det = 0;
   169	  for (int t = 0; t < trials; ++t) {
   170	    Path p = rand_path(rng, rng.range(3, 7));
   171	    uint64_t nonce = rng.nonce();
   172	    Header hdr = forward(kr, p, 0, nonce);
   173	    std::vector<Path> one{p};
   174	    ExpectationTable tb(kr, one);
   175	
   176	    Header flipped = hdr;
   177	    flipped.acc ^= 1;
   178	    ++flip_total;
   179	    if (!tb.verify_tag(flipped).ok) ++flip_det;
   180	
   181	    Path forged = rand_path(rng, rng.range(3, 7));
   182	    if (!(forged == p)) {
   183	      ++forge_total;
   184	      Header fh = forward(kr, forged, 0, nonce);
   185	      if (!tb.verify_tag(fh).ok) ++forge_det;
   186	    }
   187	  }
   188

   191	  std::printf("================================================================\n\n");
   192	  std::printf("| 编号 | 攻击手法     | 检出 / 样本 | 检出率 |\n");
   193	  std::printf("| ---- | ------------ | ----------- | ------ |\n");
   194	  for (const Row& r : rows) {

   197	                r.detected, r.total, 100.0 * r.detected / r.total);
   198	  }
   199	  std::printf("| S9   | 验证头比特翻转 | %5d/%-5d | %5.1f%% |\n", flip_det, flip_total,
   200	              100.0 * flip_det / flip_total);
   201	  std::printf("| S10  | 随机伪造验证头 | %5d/%-5d | %5.1f%% |\n", forge_det, forge_total,
   202	              100.0 * forge_det / forge_total);
   203	  std::printf("\n诚实样本接受率：%.2f%%（%d/%d）\n", 100.0 * honest_ok / trials,
   204	              honest_ok, trials);

   216	      p.push_back(Hop{2, 1, 2});

   220	      x /= 11;
   221	      p.push_back(Hop{4 + (x % 11), 1, 3});
   222	      p.push_back(Hop{15, 1, 2});
   223	      p.push_back(Hop{16, 1, 2});
   224	      cands.push_back(p);

   247	    std::vector<Path> cands;

     3	// 拓扑与场景定义见 topo.cpp。五轮：
     4	//   R1 正常        转发表与候选路径一致 → 双向全部通过，流级窗口全过
     5	//   R2 标签错配    s1 把 tag0 改走 P1 分支（数据面自洽）→ 正向全失败，

    10	//                  诚实包不受影响（fail = 0）
    11	//   R5 流级压缩    正常流量 + CFMV：出口按流聚合上报，控制器每窗口一次判定，

   232	    print_dir("正向", 0, ctl);

   234	    const DirStat& f = ctl.stat(0);
   235	    const DirStat& r = ctl.stat(1);
   236	    results.push_back(check("R6", f.packets > 0 && f.fail == 0 && r.pass > 0 &&
   237	                                      r.fail == 0,
   238	                            "故障切换后流量应全部走 P1 且验证零误报"));
   239	    (void)evs;
   240	  }
   241	
   242	  // ------------------------------------------------------------ R7 多重故障
   243	  {
   244	    // 双故障：s1 标签错配（改道 P1）+ s3 比特翻转。定位应同时报出 {s1, s3}：
   245	    // s1 = 改道/改端口嫌疑（位置偏离），s3 = 篡改嫌疑（观测值偏离）
   246	    Controller ctl(SEED, directions());
   247	    auto evs = run_round("R7 多重故障（改道+改头）", Scenario::kTagflip, "none",
   248	                         "bitflip", PINGS, ctl);
   249	    print_dir("正向", 0, ctl);
   250	    print_dir("反向", 1, ctl);
   251	    const DirStat& f = ctl.stat(0);
   252	    const DirStat& r = ctl.stat(1);
   253	    // 汇总正向全部失败事件的嫌疑集合
   254	    bool has1 = false, has3 = false;
   255	    for (const Event& e : evs)
   256	      if (e.mac == directions()[0].mac && !e.ok && !e.replay)
   257	        for (int sid : e.suspects) {
   258	          if (sid == 1) has1 = true;
   259	          if (sid == 3) has3 = true;
   260	        }
   261	    std::printf("      → 定位：嫌疑集合 {%s%s}\n", has1 ? "s1" : "",
   262	                has1 && has3 ? ", s3" : has3 ? "s3" : "");
   263	    results.push_back(check("R7", f.packets > 0 && f.fail == f.packets &&
   264	                                      r.fail == 0 && has1 && has3,
   265	                            "双故障应全检出且同时定位 s1 与 s3，反向不受影响"));
   266	  }
   267	
   268	  // ------------------------------------------------------------ 汇总
   269	  std::printf("\n----------------------------------------------------------------\n");
   270	  int bad = 0;
   271	  for (const RoundResult& r : results) {
   272	    std::printf("%s  %s\n", r.ok ? "✅" : "❌", r.name.c_str());
   273	    if (!r.ok) {
   274	      std::printf("     %s\n", r.detail.c_str());
   275	      ++bad;
   276	    }
   277	  }
   278	  if (bad) {
   279	    std::printf("结论：%d 轮未通过\n", bad);
   280	    return 1;
   281	  }
   282	  std::printf("结论：R1–R7 全部通过 —— 端到端验证成立\n");
   283	  return 0;
   284	}

   125	struct Tailer {
   126	  std::string path;
   127	  FILE* f = nullptr;
   128	  long off = 0;
   129	  std::string rest;
   130	
   131	  void close() {
   132	    if (f) std::fclose(f);
   133	    f = nullptr;
   134	    off = 0;
   135	    rest.clear();
   136	  }
   137	
   138	  // 把新增的完整行交给 fn；文件被截断时自动重开
   139	  void pump(const std::function<void(const std::string&)>& fn) {
   140	    struct stat st {};
   141	    if (::stat(path.c_str(), &st) != 0) return;
   142	    if (!f) {
   143	      f = std::fopen(path.c_str(), "r");
   144	      if (!f) return;
   145	      off = 0;
   146	      rest.clear();
   147	    }
   148	    if ((long)st.st_size < off) {  // 被清空重写
   149	      std::fclose(f);
   150	      f = std::fopen(path.c_str(), "r");
   151	      if (!f) return;
   152	      off = 0;
   153	      rest.clear();
   154	    }
   155	    std::fseek(f, off, SEEK_SET);
   156	    char buf[65536];
   157	    size_t n = std::fread(buf, 1, sizeof(buf), f);
   158	    if (n == 0) return;
   159	    off += (long)n;
   160	    std::string data = rest + std::string(buf, n);
   161	    size_t start = 0;
   162	    while (true) {
   163	      size_t nl = data.find('\n', start);
   164	      if (nl == std::string::npos) break;
   165	      std::string line = data.substr(start, nl - start);
   166	      start = nl + 1;
   167	      if (!line.empty() && line.back() == '\r') line.pop_back();
   168	      if (!line.empty()) fn(line);
   169	    }
   170	    rest = data.substr(start);
   171	    if (rest.size() > (1 << 20)) rest.clear();  // 防御：异常长行丢弃
   172	  }
   173	};

   191	void handle_line(const std::string& line) {
   192	  // 黑洞检测的 nonce 集合（正向 h1→h2）：TRACE mac sid in out tag nonce pos v
   193	  {
   194	    unsigned a, b, c, d;
   195	    unsigned long long nonce;
   196	    char mac[32];
   197	    if (std::sscanf(line.c_str(), "TRACE %31s %u %u %u %u %llu", mac, &a, &b, &c,
   198	                    &d, &nonce) == 6 &&
   199	        std::strcmp(mac, H2MAC) == 0)
   200	      g_state.traced_h2.insert(nonce);
   201	  }
   202	  Event ev = g_ctl->feed(line);
   203	  if (ev.kind.empty()) return;
   204	
   205	  if (ev.kind == "packet" && ev.mac == H2MAC)
   206	    g_state.reported_h2.insert(ev.nonce);
   207	
   208	  int d = dir_index(ev.mac);
   209	  if (d < 0) return;
   210	  DirStat& st = g_state.stats[(size_t)d];
   211	  DirStat now = g_ctl->stat((size_t)d);
   212	  st = now;
   213	
   214	  EventView v;
   215	  v.t = mono_s() - g_started;
   216	  v.mac = ev.mac;
   217	  char nh[20];
   218	  std::snprintf(nh, sizeof(nh), "0x%014llx", (unsigned long long)ev.nonce);
   219	  v.nonce_hex = nh;
   220	  v.tag = (int)ev.tag;
   221	  v.ok = ev.ok;
   222	  v.replay = ev.replay;
   223	  v.matched = ev.matched;
   224	  v.suspect = ev.suspect_sid;
   225	  v.suspects = ev.suspects;
   226	  v.lost = false;
   227	  v.kind = ev.kind;
   228	  v.n = ev.n;
   229	  v.us = ev.verify_us;
   230	  g_state.us_sum[(size_t)d] += ev.verify_us;
   231	  g_state.us_cnt[(size_t)d] += 1;
   232	  if (ev.suspect_sid >= 0) {
   233	    g_state.suspect = ev.suspect_sid;
   234	    for (int sid : ev.suspects)
   235	      if (g_state.suspects.empty() || g_state.suspects.back() != sid)
   236	        g_state.suspects.push_back(sid);
   237	  }
   238	  push_event(v);
   239	}

   336	void serve(int fd) {
   337	  // 连接级防护：读 2s / 写 5s 超时，防止空闲预连接泄漏线程
   338	  struct timeval rv{2, 0}, sv{5, 0};
   339	  ::setsockopt(fd, SOL_SOCKET, SO_RCVTIMEO, &rv, sizeof(rv));
   340	  ::setsockopt(fd, SOL_SOCKET, SO_SNDTIMEO, &sv, sizeof(sv));
   341	  int one = 1;
   342	  ::setsockopt(fd, IPPROTO_TCP, TCP_NODELAY, &one, sizeof(one));
   343	
   344	  std::string req;
   345	  char buf[4096];
   346	  // 读完整个头部（\r\n\r\n），再按 Content-Length 收完 body
   347	  while (req.find("\r\n\r\n") == std::string::npos && req.size() < 65536) {
   348	    ssize_t n = ::recv(fd, buf, sizeof(buf), 0);
   349	    if (n <= 0) break;
   350	    req.append(buf, (size_t)n);
   351	  }
   352	  if (req.find("\r\n\r\n") == std::string::npos) {
   353	    ::close(fd);
   354	    return;
   355	  }
   356	  size_t content_len = 0;
   357	  auto cl = req.find("Content-Length: ");
   358	  if (cl == std::string::npos) cl = req.find("Content-length: ");
   359	  if (cl != std::string::npos)
   360	    content_len = (size_t)std::atoi(req.c_str() + cl + 15);
   361	  auto body_start = req.find("\r\n\r\n");
   362	  while (content_len > 0 && req.size() - body_start - 4 < content_len &&
   363	         req.size() < 1 << 20) {
   364	    ssize_t n = ::recv(fd, buf, sizeof(buf), 0);
   365	    if (n <= 0) break;
   366	    req.append(buf, (size_t)n);
   367	  }
   368	  std::string body =
   369	      content_len ? req.substr(body_start + 4, content_len) : "";
   370	
   371	  std::istringstream ss(req.substr(0, req.find("\r\n")));
   372	  std::string method, path, ver;
   373	  ss >> method >> path >> ver;
   374	
   375	  std::string resp;
   376	  if (method == "GET" && (path == "/" || path == "/index.html"))
   377	    resp = http_get(fd, "/");
   378	  else if (method == "GET" && path == "/api/state")
   379	    resp = http_get(fd, path);
   380	  else if (method == "POST" && path == "/api/scenario")
   381	    resp = http_post(fd, path, body);
   382	  else if (method == "POST" && path == "/api/traffic")
   383	    resp = http_post(fd, path, body);
   384	  else if (method == "POST" && path == "/api/reset")
   385	    resp = http_post(fd, path, body);
   386	  else {
   387	    resp = "HTTP/1.1 404 Not Found\r\nConnection: close\r\nContent-Length: 0\r\n\r\n";
   388	  }
   389	  // 显式声明短连接：避免浏览器复用已被服务端关闭的陈旧连接（NAT 下会挂 20s）
   390	  auto cpos = resp.find("\r\n\r\n");
   391	  resp.insert(cpos, "\r\nConnection: close");
   392	  size_t sent = 0;
   393	  while (sent < resp.size()) {
   394	    ssize_t w = ::send(fd, resp.data() + sent, resp.size() - sent, 0);
   395	    if (w <= 0) break;
   396	    sent += (size_t)w;
   397	  }
   398	  ::shutdown(fd, SHUT_WR);
   399	  // 收掉残余字节让对端干净关闭
   400	  while (::recv(fd, buf, sizeof(buf), 0) > 0) {
   401	  }
   402	  ::close(fd);
   403	}

   405	std::string esc(const std::string& s) {
   406	  std::string o;
   407	  for (char c : s) {
   408	    switch (c) {
   409	      case '"': o += "\\\""; break;
   410	      case '\\': o += "\\\\"; break;
   411	      case '\n': o += "\\n"; break;
   412	      default:
   413	        if ((unsigned char)c < 0x20) o += ' ';
   414	        else o += c;
   415	    }
   416	  }
   417	  return o;
   418	}

   477	std::string find_in_body(const std::string& body, const std::string& key) {
   478	  auto k = body.find("\"" + key + "\"");
   479	  if (k == std::string::npos) return "";
   480	  auto colon = body.find(':', k);
   481	  if (colon == std::string::npos) return "";
   482	  auto q1 = body.find('"', colon);
   483	  if (q1 == std::string::npos) return "";
   484	  auto q2 = body.find('"', q1 + 1);
   485	  if (q2 == std::string::npos) return "";
   486	  return body.substr(q1 + 1, q2 - q1 - 1);
   487	}

   489	std::string http_post(int, const std::string& path, const std::string& body) {
   490	  if (path == "/api/scenario") {
   491	    std::string name = find_in_body(body, "name");
   492	    bool known = false;
   493	    for (const char* s : kScenarios)
   494	      if (name == s) known = true;
   495	    if (!known)
   496	      return "HTTP/1.1 400 Bad Request\r\nContent-Length: 0\r\n\r\n";
   497	    stop_switches();
   498	    reset_run(name);
   499	    write_tables(scenario_tables(name));
   500	    apply_link_state(scenario_tables(name));
   501	    start_switches(scenario_evil_s2(name), scenario_evil_s3(name));
   502	    std::this_thread::sleep_for(std::chrono::milliseconds(700));  // 等表热加载
   503	    return "HTTP/1.1 200 OK\r\nContent-Length: 2\r\n\r\nok";
   504	  }
   505	  if (path == "/api/traffic") {
   506	    int count = 20;
   507	    {
   508	      std::lock_guard<std::mutex> lk(g_state.mu);
   509	      auto k = body.find("\"count\"");
   510	      if (k != std::string::npos) {
   511	        auto colon = body.find(':', k);
   512	        if (colon != std::string::npos) {
   513	          int v = std::atoi(body.c_str() + colon + 1);
   514	          if (v > 0 && v <= 500) count = v;
   515	        }
   516	      }
   517	      g_state.traffic_packets += (uint64_t)count;
   518	    }
   519	    // 异步执行：POST 立即返回，避免占用连接线程（ping 最多几秒）
   520	    bool expect = false;
   521	    if (g_traffic_running.compare_exchange_strong(expect, true))
   522	      std::thread(traffic_worker, count).detach();
   523	    return "HTTP/1.1 200 OK\r\nContent-Length: 2\r\n\r\nok";
   524	  }
   525	  if (path == "/api/reset") {
   526	    std::string sc;
   527	    {

   530	    }
   531	    reset_run(sc);
   532	    return "HTTP/1.1 200 OK\r\nContent-Length: 2\r\n\r\nok";
   533	  }
   534	  return "HTTP/1.1 404 Not Found\r\nContent-Length: 0\r\n\r\n";

   141	struct Rng {
   142	  std::mt19937_64 g;
   143	  explicit Rng(uint64_t seed) : g(seed) {}
   144	  uint64_t next() { return g(); }
   145	  uint32_t range(uint32_t lo, uint32_t hi) { return lo + (uint32_t)(g() % (uint64_t)(hi - lo + 1)); }
   146	  uint64_t nonce() { return g() & ((1ULL << 56) - 1); }

   149	Path rand_path(Rng& r, uint32_t n_hops) {
   150	  Path p;
   151	  p.push_back(Hop{1, 1, 2});
   152	  std::vector<uint32_t> pool;
   153	  for (uint32_t s = 2; s <= 15; ++s) pool.push_back(s);
   154	  for (size_t i = pool.size(); i > 1; --i) {
   155	    size_t j = r.next() % i;
   156	    std::swap(pool[i - 1], pool[j]);
   157	  }
   158	  for (uint32_t i = 0; i + 1 < n_hops; ++i)
   159	    p.push_back(Hop{pool[i], 1, (uint32_t)(2 + (i % 2))});
   160	  p.push_back(Hop{16, 1, 2});
   161	  return p;
   162	}

   173	int exp_attack_matrix(int trials) {
   174	  g_report << "\n## [1] 十类攻击检出矩阵（每组 " << trials << " 次，64 位模域）\n\n";
   175	  g_report << "| 编号 | 攻击手法 | 检出/样本 | 检出率 |\n|---|---|---|---|\n";
   176	  Rng rng(20260501);
   177	  KeyRing kr = KeyRing::generate(16, 4, 20260501);
   178	  struct Row { const char* code; const char* name; int det = 0, tot = 0; };
   179	  std::vector<Row> rows = {{"S1","改出端口"},{"S2","改入端口"},{"S3","交换机旁路"},
   180	                           {"S4","跳序交换"},{"S5","删除一跳"},{"S6","插入一跳"},
   181	                           {"S7","整条替换"},{"S8","截断路径"}};
   182	  int honest = 0;
   183	  for (int t = 0; t < trials; ++t) {
   184	    Path p = rand_path(rng, rng.range(3, 7));
   185	    Header h = forward(kr, p, 0, rng.nonce());
   186	    std::vector<Path> one{p};
   187	    ExpectationTable tb(kr, one);
   188	    if (tb.verify_tag(h).ok) ++honest;
   189	    Path bad[8] = {atk_out_port(p, rng), atk_in_port(p, rng), atk_bypass(p, rng),
   190	                   atk_reorder(p, rng), atk_delete(p, rng), atk_insert(p, rng),
   191	                   atk_replace(p, rng), atk_truncate(p, rng)};
   192	    for (int a = 0; a < 8; ++a) {
   193	      if (bad[a] == p) continue;
   194	      ++rows[a].tot;
   195	      Header bh = forward(kr, bad[a], 0, rng.nonce());
   196	      if (!tb.verify_tag(bh).ok) ++rows[a].det;
   197	    }
   198	  }
   199	  int flip_t = 0, flip_d = 0, forge_t = 0, forge_d = 0;
   200	  for (int t = 0; t < trials; ++t) {
   201	    Path p = rand_path(rng, rng.range(3, 7));
   202	    Header h = forward(kr, p, 0, rng.nonce());
   203	    std::vector<Path> one{p};
   204	    ExpectationTable tb(kr, one);
   205	    Header f = h; f.acc ^= 1;
   206	    ++flip_t; if (!tb.verify_tag(f).ok) ++flip_d;
   207	    Path g = rand_path(rng, rng.range(3, 7));
   208	    if (!(g == p)) { ++forge_t; Header gh = forward(kr, g, 0, rng.nonce()); if (!tb.verify_tag(gh).ok) ++forge_d; }
   209	  }
   210	  for (auto& r : rows)
   211	    g_report << "| " << r.code << " | " << r.name << " | " << r.det << "/" << r.tot
   212	             << " | " << (r.tot ? 100.0 * r.det / r.tot : 0) << "% |\n";

   462	// ---------------------------------------------------------------- [9] 定位成功率
   463	struct HopRec {
   464	  uint32_t sid, in, out, pos;
   465	  uint64_t v;
   466	};
   467	
   468	// 沿实际路径逐跳生成取证记录（与 vswitch --trace 同口径：pos 1 基），
   469	// tamper_hop >= 0 时该跳在合法贡献之后翻转 1 bit（恶意改头）。
   470	std::vector<HopRec> trace_of(const KeyRing& kr, const Path& p, uint64_t nonce,
   471	                             int tamper_hop) {
   472	  std::vector<HopRec> recs;
   473	  uint64_t acc = 0;
   474	  for (size_t i = 0; i < p.size(); ++i) {
   475	    acc = addmod(acc, mulmod(contribution(kr, p[i], h_of(nonce)),
   476	                             powmod(RHO, (uint64_t)i)));
   477	    if ((int)i == tamper_hop) acc ^= 1;  // 篡改后下游在篡改值上继续（与 vswitch 一致）
   478	    recs.push_back({p[i].sid, p[i].in_port, p[i].out_port, (uint32_t)(i + 1), acc});
   479	  }
   480	  return recs;
   481	}
   482	
   483	size_t first_div(const Path& a, const Path& b) {
   484	  size_t i = 0;
   485	  while (i < a.size() && i < b.size() && a[i] == b[i]) ++i;
   486	  return i;
   487	}

   489	void exp_localization(int trials) {
   490	  g_report << "\n## [9] 异常交换机定位成功率（随机路径 5–9 跳，随机故障位置）\n\n";
   491	  Rng rng(777);
   492	  const KeyRing kr = KeyRing::generate(16, 4, SEED);
   493	  const char* MAC = "00:00:00:00:00:02";
   494	
   495	  // 主实验：单候选集 + 逐跳取证。四类样本：单篡改 / 单改道 / 双故障 / 诚实对照。
   496	  // 真值口径：篡改跳 = 实际路径该跳 sid；改道跳 = 声称路径首分歧跳 sid（与
   497	  // 控制器定位语义一致）。真攻击纪律：bad == claimed 的样本不计入。
   498	  int tam_n = 0, tam_hit = 0, dev_n = 0, dev_hit = 0, two_n = 0, two_hit = 0;
   499	  int ho_n = 0, ho_ok = 0, overmark = 0;
   500	  int exact_all = 0, fault_all = 0;
   501	  for (int t = 0; t < trials; ++t) {
   502	    Path p = rand_path(rng, rng.range(4, 8));  // 总跳数 5–9
   503	    uint64_t nonce = rng.nonce();
   504	    uint32_t mode = (uint32_t)(rng.next() % 4);
   505	    Path bad = p;
   506	    int tam = -1;
   507	    bool dev = false;
   508	    if (mode == 0 || mode == 2) tam = (int)rng.range(0, (uint32_t)p.size() - 1);
   509	    if (mode == 1 || mode == 2) {
   510	      for (int tries = 0; tries < 10 && !dev; ++tries) {
   511	        bad = (rng.next() % 2) ? atk_out_port(p, rng) : atk_bypass(p, rng);
   512	        dev = !(bad == p);
   513	      }
   514	      if (!dev) continue;
   515	    }
   516	    Controller c(SEED, {Direction{MAC, {p}}});
   517	    auto recs = trace_of(kr, bad, nonce, tam);
   518	    char buf[192];
   519	    for (const HopRec& r : recs) {
   520	      std::snprintf(buf, sizeof(buf), "TRACE %s %u %u %u 0 %llu %u %llu", MAC,
   521	                    r.sid, r.in, r.out, (unsigned long long)nonce, r.pos,
   522	                    (unsigned long long)r.v);
   523	      c.feed(buf);
   524	    }
   525	    std::snprintf(buf, sizeof(buf), "4 0 %llu %llu %s",
   526	                  (unsigned long long)nonce, (unsigned long long)recs.back().v, MAC);
   527	    Event ev = c.feed(buf);
   528	
   529	    if (!dev && tam < 0) {  // 诚实样本：应判定通过且无嫌疑
   530	      ++ho_n;
   531	      if (ev.ok && ev.suspects.empty()) ++ho_ok;
   532	      continue;
   533	    }
   534	    std::vector<int> truth;
   535	    if (tam >= 0) truth.push_back((int)bad[tam].sid);
   536	    if (dev) {
   537	      size_t fd = first_div(p, bad);
   538	      if (fd >= p.size()) fd = p.size() - 1;
   539	      truth.push_back((int)p[fd].sid);
   540	    }
   541	    std::sort(truth.begin(), truth.end());
   542	    truth.erase(std::unique(truth.begin(), truth.end()), truth.end());
   543	    std::vector<int> sus = ev.suspects;
   544	    std::sort(sus.begin(), sus.end());
   545	    bool hit = truth == sus;
   546	    bool over = false;
   547	    for (int s : sus)
   548	      if (!std::binary_search(truth.begin(), truth.end(), s)) over = true;
   549	    if (over) ++overmark;
   550	    ++fault_all;
   551	    if (hit) ++exact_all;
   552	    if (dev && tam < 0) { ++dev_n; if (hit) ++dev_hit; }
   553	    else if (!dev && tam >= 0) { ++tam_n; if (hit) ++tam_hit; }
   554	    else { ++two_n; if (hit) ++two_hit; }
   555	  }
   556	  g_report << "| 故障类型 | 样本数 | 嫌疑集精确命中 | 精确率 |\n|---|---|---|---|\n";
   557	  g_report << "| 恶意改头（S9） | " << tam_n << " | " << tam_hit << " | "
   558	           << (tam_n ? 100.0 * tam_hit / tam_n : 0) << "% |\n";
   559	  g_report << "| 路径改道（S1/S3） | " << dev_n << " | " << dev_hit << " | "
   560	           << (dev_n ? 100.0 * dev_hit / dev_n : 0) << "% |\n";
   561	  g_report << "| 双故障（改头+改道） | " << two_n << " | " << two_hit << " | "
   562	           << (two_n ? 100.0 * two_hit / two_n : 0) << "% |\n";
   563	  g_report << "| 诚实对照（误报） | " << ho_n << " | 误报 " << (ho_n - ho_ok)
   564	           << " | " << (ho_n ? 100.0 * ho_ok / ho_n : 0) << "% |\n";
   565	  g_report << "- 全部故障样本精确命中率：" << exact_all << "/" << fault_all << "（"
   566	           << (fault_all ? 100.0 * exact_all / fault_all : 0) << "%），"
   567	           << "含诚实交换机误标样本 " << overmark << " 个\n";
   568	
   569	  // 对照：k=4 候选集，有取证 vs 无取证。改道到合法路径（tagflip 类）两种都
   570	  // 能定位；集合外篡改只有取证能定位——这是 --trace 取证存在的理由。
   571	  g_report << "\n### 对照：有逐跳取证 vs 无取证（k=4 候选集，各 "
   572	           << trials / 4 << " 次）\n\n";
   573	  std::vector<Path> cands;
   574	  for (uint32_t i = 0; i < 4; ++i) {
   575	    Path q;
   576	    q.push_back(Hop{1, 1, 2});
   577	    q.push_back(Hop{2 + i, 1, 2});
   578	    q.push_back(Hop{6 + i, 1, 3});
   579	    q.push_back(Hop{10 + i, 1, 2});
   580	    q.push_back(Hop{16, 1, 2});
   581	    cands.push_back(q);
   582	  }
   583	  int tf[2] = {0, 0}, tf_n = 0, bf[2] = {0, 0}, bf_n = 0;
   584	  for (int t = 0; t < trials / 2; ++t) {
   585	    uint64_t nonce = rng.nonce();
   586	    // B1 改道到合法路径：沿 cands[j] 转发却声称 tag 0。两种口径各自有定义：
   587	    //   有取证 = 声称路径首分歧跳；无取证 = 候选对比的最后公共交换机（路由分歧点）
   588	    {
   589	      uint32_t j = rng.range(1, 3);
   590	      Header h = forward(kr, cands[j], 0, nonce);
   591	      size_t fd = first_div(cands[0], cands[j]);
   592	      size_t common = 0;
   593	      while (common < cands[0].size() && common < cands[j].size() &&
   594	             cands[0][common].sid == cands[j][common].sid)
   595	        ++common;
   596	      int want[2] = {(int)cands[0][common > 0 ? common - 1 : 0].sid,
   597	                     (int)cands[0][fd].sid};
   598	      for (int with_trace = 0; with_trace < 2; ++with_trace) {
   599	        Controller c(SEED, {Direction{MAC, cands}});
   600	        char buf[192];
   601	        if (with_trace) {
   602	          for (const HopRec& r : trace_of(kr, cands[j], nonce, -1)) {
   603	            std::snprintf(buf, sizeof(buf), "TRACE %s %u %u %u 0 %llu %u %llu", MAC,
   604	                          r.sid, r.in, r.out, (unsigned long long)nonce, r.pos,
   605	                          (unsigned long long)r.v);
   606	            c.feed(buf);
   607	          }
   608	        }
   609	        std::snprintf(buf, sizeof(buf), "4 0 %llu %llu %s",
   610	                      (unsigned long long)nonce, (unsigned long long)h.acc, MAC);
   611	        Event ev = c.feed(buf);
   612	        if (!ev.ok && ev.suspects.size() == 1 && ev.suspects[0] == want[with_trace])
   613	          ++tf[with_trace];
   614	      }
   615	      ++tf_n;
   616	    }
   617	    // B2 集合外篡改：cands[0] 改一跳出端口，不在任何候选里
   618	    {
   619	      Path bad;
   620	      for (int tries = 0; tries < 10; ++tries) {
   621	        bad = atk_out_port(cands[0], rng);
   622	        if (!(bad == cands[0])) break;
   623	      }
   624	      if (!(bad == cands[0])) {
   625	        uint64_t n2 = rng.nonce();
   626	        Header h = forward(kr, bad, 0, n2);
   627	        for (int with_trace = 0; with_trace < 2; ++with_trace) {
   628	          Controller c(SEED, {Direction{MAC, cands}});
   629	          if (with_trace) {
   630	            char buf[192];
   631	            for (const HopRec& r : trace_of(kr, bad, n2, -1)) {
   632	              std::snprintf(buf, sizeof(buf), "TRACE %s %u %u %u 0 %llu %u %llu", MAC,
   633	                            r.sid, r.in, r.out, (unsigned long long)n2, r.pos,
   634	                            (unsigned long long)r.v);
   635	              c.feed(buf);
   636	            }
   637	          }
   638	          char buf[128];
   639	          std::snprintf(buf, sizeof(buf), "4 0 %llu %llu %s",
   640	                        (unsigned long long)n2, (unsigned long long)h.acc, MAC);
   641	          Event ev = c.feed(buf);
   642	          size_t fd = first_div(cands[0], bad);
   643	          if (!ev.ok && ev.suspects.size() == 1 &&
   644	              ev.suspects[0] == (int)cands[0][fd].sid)
   645	            ++bf[with_trace];
   646	        }
   647	        ++bf_n;
   648	      }
   649	    }
   650	  }
   651	  g_report << "| 故障 | 有取证定位 | 无取证定位 |\n|---|---|---|\n";
   652	  g_report << "| 改道到合法路径（tagflip） | " << tf[1] << "/" << tf_n << "（"
   653	           << (tf_n ? 100.0 * tf[1] / tf_n : 0) << "%） | " << tf[0] << "/" << tf_n
   654	           << "（" << (tf_n ? 100.0 * tf[0] / tf_n : 0) << "%） |\n";
   655	  g_report << "| 集合外篡改（改出端口） | " << bf[1] << "/" << bf_n << "（"
   656	           << (bf_n ? 100.0 * bf[1] / bf_n : 0) << "%） | " << bf[0] << "/" << bf_n
   657	           << "（" << (bf_n ? 100.0 * bf[0] / bf_n : 0) << "%） |\n";
   658	  g_report << "- 结论：合法集合内的改道可由候选对比定位（取证非必需）；"
   659	              "集合外篡改只有逐跳取证能定位——取证开销换的是闭集之外的可定位性。\n";
   660	}

   662	// ---------------------------------------------------------------- [10] 基线对比
   663	// 基线 A（Icing 类）：逐跳 MAC 链。每跳算 8B tag 并追加进验证头，控制器对
   664	// 候选路径重放整条链。与 MPV-2 共用同一密钥体系（端口密钥），同测试床公平。
   665	uint64_t mac_hop(const KeyRing& kr, const Hop& h, uint64_t prev) {
   666	  uint8_t k[32] = {0}, buf[20];
   667	  uint64_t ks = kr.port_key(h);
   668	  for (int i = 0; i < 8; ++i) k[i] = (uint8_t)((ks >> (8 * (7 - i))) & 0xff);
   669	  for (int i = 0; i < 8; ++i) buf[i] = (uint8_t)((prev >> (8 * (7 - i))) & 0xff);
   670	  uint32_t f[3] = {h.sid, h.in_port, h.out_port};
   671	  for (int j = 0; j < 3; ++j)
   672	    for (int i = 0; i < 4; ++i) buf[8 + 4 * j + i] = (uint8_t)((f[j] >> (8 * (3 - i))) & 0xff);
   673	  uint8_t out[8];
   674	  blake2b(out, 8, buf, sizeof(buf), k, 32);
   675	  uint64_t v = 0;
   676	  for (int i = 0; i < 8; ++i) v = (v << 8) | out[i];
   677	  return v;
   678	}
   679	
   680	std::vector<uint64_t> mac_chain(const KeyRing& kr, const Path& p, uint64_t nonce) {
   681	  std::vector<uint64_t> tags;
   682	  uint64_t prev = nonce;
   683	  for (const Hop& h : p) {
   684	    prev = mac_hop(kr, h, prev);
   685	    tags.push_back(prev);
   686	  }
   687	  return tags;
   688	}
   689	
   690	bool mac_verify(const KeyRing& kr, const Path& p, uint64_t nonce,
   691	                const std::vector<uint64_t>& tags) {
   692	  std::vector<uint64_t> rec = mac_chain(kr, p, nonce);
   693	  return rec == tags;
   694	}

   696	// 基线 B（Atlas 类）：概率采样多路径验证。包以概率 ρ 被验证，检出率 ≈ ρ；
   697	// 控制器负载 = ρ。MPV-2+CFMV 的负载 = 1/W（W 为流级窗口）且全检出。
   698	void exp_baseline(int trials) {
   699	  g_report << "\n## [10] 算法级基线对比（同测试床：同密钥体系、同攻击样本）\n\n";
   700	
   701	  // A：逐跳 MAC 链判定时延 vs k（h=7，中位数）
   702	  g_report << "### 逐跳 MAC 链（Icing 类）vs MPV-2：控制器判定时延 vs k（h=7）\n\n";
   703	  g_report << "| k | MAC 链逐候选 (µs) | MPV-2 tag (µs) | MPV-2 scan (µs) |\n|---|---|---|---|\n";
   704	  Rng rng(31);
   705	  KeyRing kr = KeyRing::generate(16, 4, 20260501);
   706	  for (uint32_t k : {1u, 2u, 4u, 8u, 16u, 32u, 64u}) {
   707	    std::vector<Path> cands;
   708	    for (uint32_t i = 0; i < k; ++i) {
   709	      Path p;
   710	      p.push_back(Hop{1, 1, 2});
   711	      p.push_back(Hop{2, 1, 2});
   712	      p.push_back(Hop{3, 1, 2});
   713	      uint32_t x = i;
   714	      p.push_back(Hop{4 + (x % 11), 1, 2}); x /= 11;
   715	      p.push_back(Hop{4 + (x % 11), 1, 3});
   716	      p.push_back(Hop{15, 1, 2});
   717	      p.push_back(Hop{16, 1, 2});
   718	      cands.push_back(p);
   719	    }
   720	    Header h = forward(kr, cands[k - 1], (uint8_t)((k - 1) & 0xff), rng.nonce());
   721	    std::vector<uint64_t> tags = mac_chain(kr, cands[k - 1], h.nonce);
   722	    ExpectationTable tb(kr, cands);
   723	    std::vector<double> tm, tt, ts;
   724	    for (int i = 0; i < 2000; ++i) {
   725	      double t0 = now_us();
   726	      for (uint32_t c = 0; c < k; ++c) mac_verify(kr, cands[c], h.nonce, tags);
   727	      tm.push_back(now_us() - t0);
   728	      t0 = now_us(); tb.verify_tag(h); tt.push_back(now_us() - t0);
   729	      t0 = now_us(); tb.verify_scan(h); ts.push_back(now_us() - t0);
   730	    }
   731	    g_report << "| " << k << " | " << median(tm) << " | " << median(tt) << " | "
   732	             << median(ts) << " |\n";
   733	  }
   734	  g_report << "- MAC 链每包重放 O(k·h) 次哈希、随 k 线性增长；MPV-2 tag 模式 O(1) "
   735	              "与 k 无关。MAC 链的验证头为逐跳追加式（8B/跳，h=7 时 63B），头长对比见 [6]。\n";
   736	
   737	  // B：概率采样（Atlas 类）：检出率 vs 判定负载。攻击包数 = trials（与 [1] 的
   738	  // 攻击样本数同义），总包数 = 10 × trials，攻击占比 10%。
   739	  g_report << "\n### 概率采样多路径（Atlas 类）vs MPV-2+CFMV：漏检率 vs 控制器负载\n\n";
   740	  const int NPKT = trials * 10;
   741	  Rng rng2(77);
   742	  Path p = rand_path(rng2, 7);
   743	  ExpectationTable tb(kr, std::vector<Path>{p});
   744	  struct Pkt { uint64_t nonce, acc; bool atk; };
   745	  std::vector<Pkt> pkts;
   746	  pkts.reserve(NPKT);
   747	  int atk_tot = 0;
   748	  for (int i = 0; i < NPKT; ++i) {
   749	    bool atk = (i % 10) == 0;  // 10% 攻击包
   750	    uint64_t nonce = rng2.nonce();
   751	    Path q = p;
   752	    if (atk) {
   753	      ++atk_tot;
   754	      uint32_t hi = rng2.range(1, (uint32_t)p.size() - 2);
   755	      q[hi].out_port = (q[hi].out_port % 3) + 1;
   756	      if (q == p) { q[hi].out_port = q[hi].out_port == 1 ? 2 : 1; }
   757	    }
   758	    Header h = forward(kr, q, 0, nonce);
   759	    pkts.push_back({nonce, h.acc, atk});
   760	  }
   761	  g_report << "| 采样率 ρ | 控制器判定负载 | 攻击检出率 | 漏检率 |\n|---|---|---|---|\n";
   762	  std::vector<std::pair<double, double>> sample_pts;
   763	  for (double rho : {0.01, 0.05, 0.1, 0.25, 0.5, 1.0}) {
   764	    uint64_t det = 0, verified = 0;
   765	    for (const Pkt& pk : pkts) {
   766	      bool sampled = (h_of(pk.nonce) % 10000) < (uint64_t)(rho * 10000.0);
   767	      if (!sampled) continue;
   768	      ++verified;
   769	      Header h{pk.nonce, pk.acc, 0};
   770	      if (pk.atk && !tb.verify_tag(h).ok) ++det;
   771	    }
   772	    double det_rate = atk_tot ? 100.0 * det / atk_tot : 0;
   773	    g_report << "| " << rho << " | " << (100.0 * verified / NPKT) << "% | "
   774	             << det_rate << "% | " << (100 - det_rate) << "% |\n";
   775	    sample_pts.push_back({rho, 100 - det_rate});
   776	  }
   777	  g_report << "| MPV-2+CFMV（W=64） | " << (100.0 / 64) << "% | 100% | 0% |\n";
   778	  g_report << "- 采样式以漏检换负载：负载降到 1/" << 64 << " 时漏检 "
   779	           << (100 - 100.0 / 64) << "%；MPV-2+CFMV 用流级聚合把负载降到同一量级且"
   780	              "全检出（篡改包使所在窗口失衡，[1] 与 [4] 已实测检出与诚实零误报）。\n";
   781	  std::vector<std::pair<double, double>> cfmv_pts{{1.0 / 64, 0}};
   782	  std::vector<Series> ss{{"概率采样（漏检≈1−ρ）", "#fbbf24", sample_pts},
   783	                         {"MPV-2+CFMV（全检出）", "#4ade80", cfmv_pts}};
   784	  svg_chart("chart_baseline.svg", "控制器判定负载 vs 攻击漏检率（10% 攻击流量）",
   785	            "判定负载（相对逐包全验证）", "漏检率 %",
   786	            {"0.01", "0.05", "0.1", "0.25", "0.5", "1.0"},
   787	            {0.01, 0.05, 0.1, 0.25, 0.5, 1.0}, ss, 105, -3);
   788	}

   790	// ---------------------------------------------------------------- [11] 规模扩展
   791	Path path_of_hops(uint32_t hops) {
   792	  Path p;
   793	  p.push_back(Hop{1, 1, 2});
   794	  for (uint32_t i = 0; i + 2 < hops; ++i)
   795	    p.push_back(Hop{2 + (i % 14), 1, (uint32_t)(2 + (i % 2))});
   796	  p.push_back(Hop{16, 1, 2});
   797	  return p;
   798	}

   800	void exp_hops() {
   801	  g_report << "\n## [11] 规模扩展：路径跳数 h 下的开销与检出（各 2000 次）\n\n";
   802	  g_report << "| h | 数据面单跳均摊 (µs) | 期望预计算 (µs) | tag 判定 (µs) | 检出率 |\n|---|---|---|---|---|\n";
   803	  Rng rng(41);
   804	  KeyRing kr = KeyRing::generate(16, 4, 20260501);
   805	  std::vector<double> hs, pre, tag;
   806	  std::vector<std::string> ticks;
   807	  for (uint32_t h : {4u, 8u, 16u, 32u}) {
   808	    Path p = path_of_hops(h);
   809	    ExpectationTable tb(kr, std::vector<Path>{p});
   810	    // 数据面：整条 forward，均摊到每跳
   811	    std::vector<double> tf;
   812	    for (int i = 0; i < 2000; ++i) {
   813	      double t0 = now_us();
   814	      forward(kr, p, 0, rng.nonce());
   815	      tf.push_back(now_us() - t0);
   816	    }
   817	    // 控制器：期望预计算（流建立时一次）
   818	    std::vector<double> te;
   819	    for (int i = 0; i < 2000; ++i) {
   820	      double t0 = now_us();
   821	      expectation(kr, p);
   822	      te.push_back(now_us() - t0);
   823	    }
   824	    // 控制器：tag 判定（每包）
   825	    Header hh = forward(kr, p, 0, rng.nonce());
   826	    std::vector<double> tv;
   827	    for (int i = 0; i < 2000; ++i) {
   828	      double t0 = now_us();
   829	      tb.verify_tag(hh);
   830	      tv.push_back(now_us() - t0);
   831	    }
   832	    // 检出率：改端口 / 旁路 / 改头 三类均匀混合
   833	    int det = 0, tot = 0;
   834	    for (int i = 0; i < 2000; ++i) {
   835	      uint32_t kind = rng.range(0, 2);
   836	      if (kind == 2) {  // 改头：出口帧翻转一位
   837	        Header bh = forward(kr, p, 0, rng.nonce());
   838	        bh.acc ^= 1;
   839	        ++tot;
   840	        if (!tb.verify_tag(bh).ok) ++det;
   841	        continue;
   842	      }
   843	      Path bad = (kind == 0) ? atk_out_port(p, rng) : atk_bypass(p, rng);
   844	      if (bad == p) continue;
   845	      ++tot;
   846	      Header bh = forward(kr, bad, 0, rng.nonce());
   847	      if (!tb.verify_tag(bh).ok) ++det;
   848	    }
   849	    double mh = median(tf) / h, me = median(te), mt = median(tv);
   850	    hs.push_back(h); ticks.push_back(std::to_string(h));
   851	    pre.push_back(me); tag.push_back(mt);
   852	    g_report << "| " << h << " | " << mh << " | " << me << " | " << mt << " | "
   853	             << (tot ? 100.0 * det / tot : 0) << "% |\n";
   854	  }
   855	  std::vector<Series> ss{{"期望预计算（流建立一次）", "#fbbf24", {}},
   856	                         {"tag 判定（每包）", "#4ade80", {}}};
   857	  for (size_t i = 0; i < hs.size(); ++i) {
   858	    ss[0].pts.push_back({hs[i], pre[i]});
   859	    ss[1].pts.push_back({hs[i], tag[i]});
   860	  }
   861	  svg_chart("chart_hops.svg", "控制器开销 vs 路径跳数 h", "跳数 h", "µs",
   862	            ticks, hs, ss, *std::max_element(pre.begin(), pre.end()) * 1.15);
   863	  g_report << "- 数据面每跳 O(1)（均摊时延不随 h 变化，总时延线性于 h 属逐跳计算固有）"
   864	              "；控制器 tag 判定与 h 无关（期望已折叠成 A+B·h）；只有流建立时的"
   865	              "期望预计算随 h 线性——一次性成本。\n";
   866	}

   886	  std::vector<pid_t> pids;

   908	int suspect_of(const std::vector<Event>& evs, const char* mac) {
   909	  for (const Event& e : evs)
   910	    if (e.mac == mac && !e.ok && !e.replay && e.suspect_sid >= 0) return e.suspect_sid;
   911	  return -1;

   919	  {
   920	    RoundStat rs;
   921	    for (int r = 0; r < rounds; ++r) {
   922	      Controller ctl(SEED, directions());
   923	      auto evs = run_e2e_round(Scenario::kTagflip, "none", pings, ctl);
   924	      const DirStat& f = ctl.stat(0);
   925	      const DirStat& rv = ctl.stat(1);
   926	      ++rs.rounds;
   927	      if (f.packets > 0 && f.fail == f.packets) ++rs.detect_ok;
   928	      if (suspect_of(evs, H2MAC) == 1) ++rs.locate_ok;
   929	      if (rv.fail == 0) ++rs.false_pos;  // 反向不受影响计为「无误报」
   930	      // 误报列改为「反向无误报轮数」
   931	    }

   933	             << rs.locate_ok << " | " << rs.false_pos << " |\n";
   934	  }
   935	  // R3 恶意改头
   936	  {

   939	      Controller ctl(SEED, directions());
   940	      auto evs = run_e2e_round(Scenario::kNormal, "bitflip", pings, ctl);
   941	      const DirStat& f = ctl.stat(0);
   942	      const DirStat& rv = ctl.stat(1);

   160	  std::vector<Path> three{tampered};
   161	  ExpectationTable t3(kr, three);
   162	  CHECK(!t3.verify_tag(honest).ok, "端口篡改被拒绝");
   163	
   164	  // 交换机旁路
   165	  Path bypass = p;
   166	  bypass[1] = Hop{14, 1, 2};
   167	  std::vector<Path> four{bypass};
   168	  ExpectationTable t4(kr, four);
   169	  CHECK(!t4.verify_tag(honest).ok, "交换机旁路被拒绝");
   170	
   171	  // 头部比特翻转
   172	  Header flipped = honest;
   173	  flipped.acc ^= 1;
   174	  CHECK(!t1.verify_tag(flipped).ok, "验证头比特翻转被拒绝");
   175	
   176	  // 候选集定位
   177	  std::vector<Path> cands{make_path({1, 2, 5}), make_path({1, 3, 5}),
   178	                          make_path({1, 4, 5})};
   179	  ExpectationTable t5(kr, cands);
   180	  Header h2 = forward(kr, cands[1], 1, 777);
   181	  Verdict v = t5.verify_tag(h2);
   182	  CHECK(v.ok && v.matched == 1, "候选集命中唯一下标");
   183	
   184	  // tag 篡改：V 属于路径 1，tag 声称 0
   185	  Header spoof = h2;
   186	  spoof.tag = 0;
   187	  CHECK(!t5.verify_tag(spoof).ok, "tag 篡改被拒绝");
   188	  CHECK(t5.verify_scan(spoof).matched == 1, "scan 仍能识别真实路径");
   189	
   190	  // 大样本检出率
   191	  int trials = 3000, detected = 0, counted = 0;
   192	  uint64_t s = 12345;
   193	  auto next = [&s]() {
   194	    s = s * 6364136223846793005ULL + 1442695040888963407ULL;
   195	    return s;
   196	  };
   197	  for (int i = 0; i < trials; ++i) {
   198	    Path base = make_path({1, (uint32_t)(2 + next() % 8), (uint32_t)(10 + next() % 5), 16});
   199	    Header hh = forward(kr, base, 0, next() & ((1ULL << 56) - 1));
   200	    Path bad = base;
   201	    uint32_t idx = (uint32_t)(next() % bad.size());
   202	    bad[idx].out_port = 1 + (uint32_t)(next() % 4);
   203	    if (bad[idx].out_port == base[idx].out_port) bad[idx].out_port = 3;
   204	    if (bad == base) continue; // 篡改退化为无操作，不计入分母
   205	    ++counted;
   206	    std::vector<Path> tb{bad};
   207	    ExpectationTable tbad(kr, tb);
   208	    if (!tbad.verify_tag(hh).ok) ++detected;
   209	  }
   210	  CHECK_EQ(detected, counted, "随机篡改检出率 100%");
   211	}

   214	// CFMV 流级压缩 + 重放窗口

   217	static void flow_tests() {
   218	  KeyRing kr = KeyRing::generate(16, 4, 0xF10A5);
   219	  Path p = make_path({1, 6, 11, 16});
   220	  std::vector<Path> cands{p, make_path({1, 7, 12, 16})};
   221	  ExpectationTable tb(kr, cands);
   222	
   223	  // N 包聚合后一次判定通过，且窗口内每个包逐包判定也通过
   224	  FlowAgg agg;
   225	  for (uint64_t i = 0; i < 50; ++i) {
   226	    Header h = forward(kr, p, 0, 1000 + i);
   227	    agg.add(h.nonce, h.acc);
   228	    CHECK(tb.verify_tag(h).ok, "流内单包逐包判定应通过");
   229	  }
   230	  Verdict v = tb.verify_flow(0, agg.n, agg.sum_h, agg.sum_v);
   231	  CHECK(v.ok, "流级聚合判定通过");
   232	  CHECK_EQ(v.checks, 1, "流级判定检查次数为 1");
   233	
   234	  // 线性恒等式：ΣV == N·A + B·Σh（CFMV 的数学根基）
   235	  Expectation e = expectation(kr, p);
   236	  CHECK_EQ(addmod(mulmod(agg.n % P, e.A), mulmod(e.B, agg.sum_h)), agg.sum_v,
   237	           "CFMV 线性恒等式 ΣV = N·A + B·Σh");
   238	
   239	  // 聚合域篡改：动任何一个分量都会破坏恒等式
   240	  FlowAgg bad = agg;
   241	  bad.sum_v = addmod(bad.sum_v, 1);
   242	  CHECK(!tb.verify_flow(0, bad.n, bad.sum_h, bad.sum_v).ok, "聚合 sum_v 篡改被拒");
   243	  FlowAgg bad2 = agg;
   244	  bad2.sum_h = addmod(bad2.sum_h, 7);
   245	  CHECK(!tb.verify_flow(0, bad2.n, bad2.sum_h, bad2.sum_v).ok, "聚合 sum_h 篡改被拒");
   246	  CHECK(!tb.verify_flow(0, agg.n + 1, agg.sum_h, agg.sum_v).ok, "窗口包数篡改被拒");
   247	  CHECK(!tb.verify_flow(1, agg.n, agg.sum_h, agg.sum_v).ok, "流级误配候选被拒");
   248	
   249	  // 大窗口（512 包）仍然一次判定
   250	  FlowAgg big;
   251	  for (uint64_t i = 0; i < 512; ++i) {
   252	    Header h = forward(kr, p, 0, 5000 + i);
   253	    big.add(h.nonce, h.acc);
   254	  }
   255	  CHECK(tb.verify_flow(0, big.n, big.sum_h, big.sum_v).ok,
   256	        "512 包窗口一次判定通过");
   257	
   258	  // 重放窗口
   259	  NonceWindow w(64);
   260	  CHECK(w.see(12345), "新 nonce 接受");
   261	  CHECK(!w.see(12345), "重复 nonce 拒绝");
   262	  bool all_new = true;
   263	  for (uint64_t i = 0; i < 63; ++i) all_new = all_new && w.see(700 + i);
   264	  CHECK(all_new, "窗口未满时新 nonce 全部接受");
   265	  CHECK(!w.see(12345), "未滑出窗口的重复仍被拒绝");
   266	}

   268	// ---------------------------------------------------------------------------
   269	// 控制器：上报解析、重放拦截、取证定位
   270	// ---------------------------------------------------------------------------
   271	
   272	static void controller_tests() {
   273	  const char* MAC = "aa:aa:aa:aa:aa:01";
   274	  Path p0 = make_path({1, 2, 3, 4});
   275	  Path p1 = make_path({1, 5, 3, 4});
   276	  std::vector<Direction> dirs{{MAC, {p0, p1}}};
   277	  const uint64_t SEED = 999;
   278	  KeyRing kr = KeyRing::generate(16, 4, SEED);
   279	  Controller ctl(SEED, dirs);
   280	
   281	  // 诚实包：先取证后上报（与真实时序一致）
   282	  Header h = forward(kr, p0, 0, 777);
   283	  char line[160];
   284	  uint64_t hh = h_of(777);
   285	  uint64_t e = 0;
   286	  for (size_t j = 0; j < p0.size(); ++j) {
   287	    e = addmod(e, mulmod(contribution(kr, p0[j], hh), powmod(RHO, (uint64_t)j)));
   288	    std::snprintf(line, sizeof(line), "TRACE %s %u %u %u 0 777 %zu %llu", MAC,
   289	                  p0[j].sid, p0[j].in_port, p0[j].out_port, j + 1,
   290	                  (unsigned long long)e);
   291	    ctl.feed(line);
   292	  }
   293	  std::snprintf(line, sizeof(line), "4 0 %llu %llu %s",
   294	                (unsigned long long)h.nonce, (unsigned long long)h.acc, MAC);
   295	  Event ev = ctl.feed(line);
   296	  CHECK(ev.kind == "packet" && ev.ok, "控制器判定诚实包通过");
   297	  CHECK_EQ(ev.suspect_sid, -1, "诚实包无嫌疑交换机");
   298	
   299	  // 重放同一行 → nonce 窗口拦截
   300	  ev = ctl.feed(line);
   301	  CHECK(ev.replay, "重放被 nonce 窗口拦截");
   302	
   303	  // 中间篡改：s2 翻转 1 比特 → 取证定位 s2
   304	  Header bad = forward(kr, p0, 0, 888);
   305	  bad.acc ^= 1;
   306	  uint64_t h8 = h_of(888);
   307	  uint64_t e1 = contribution(kr, p0[0], h8);
   308	  uint64_t e2 = addmod(e1, mulmod(contribution(kr, p0[1], h8), RHO));
   309	  std::snprintf(line, sizeof(line), "TRACE %s 1 1 2 0 888 1 %llu", MAC,
   310	                (unsigned long long)e1);
   311	  ctl.feed(line);
   312	  std::snprintf(line, sizeof(line), "TRACE %s 2 1 3 0 888 2 %llu", MAC,
   313	                (unsigned long long)(e2 ^ 1));
   314	  ctl.feed(line);
   315	  std::snprintf(line, sizeof(line), "4 0 %llu %llu %s",
   316	                (unsigned long long)bad.nonce, (unsigned long long)bad.acc, MAC);
   317	  ev = ctl.feed(line);
   318	  CHECK(!ev.ok, "中间篡改包被拒");
   319	  CHECK_EQ(ev.suspect_sid, 2, "取证定位到 s2");
   320	
   321	  // 出口上报篡改：各跳取证全部正常但最终上报被改 → 定位到出口 s4
   322	  Header fine = forward(kr, p0, 0, 999);
   323	  hh = h_of(999);
   324	  e = 0;
   325	  for (size_t j = 0; j < p0.size(); ++j) {
   326	    e = addmod(e, mulmod(contribution(kr, p0[j], hh), powmod(RHO, (uint64_t)j)));
   327	    std::snprintf(line, sizeof(line), "TRACE %s %u %u %u 0 999 %zu %llu", MAC,
   328	                  p0[j].sid, p0[j].in_port, p0[j].out_port, j + 1,
   329	                  (unsigned long long)e);
   330	    ctl.feed(line);
   331	  }
   332	  std::snprintf(line, sizeof(line), "4 0 %llu %llu %s",
   333	                (unsigned long long)fine.nonce,
   334	                (unsigned long long)(fine.acc ^ 2), MAC);
   335	  ev = ctl.feed(line);
   336	  CHECK(!ev.ok, "出口上报篡改被拒");
   337	  CHECK_EQ(ev.suspect_sid, 4, "取证定位到出口 s4");
   338	
   339	  // 多故障：s1 改端口（改道）+ s4 篡改 —— 定位应同时报出 {1, 4}
   340	  // 实际路由 s1(1,3) → s5(1,2) → s4(2,3)，仅在 s4 处翻转 1 比特
   341	  {
   342	    const unsigned long long n7 = 1111;
   343	    uint64_t h7 = h_of(n7);
   344	    uint64_t v1 = contribution(kr, Hop{1, 1, 3}, h7);
   345	    uint64_t v2 = addmod(v1, mulmod(contribution(kr, Hop{5, 1, 2}, h7), RHO));
   346	    uint64_t v3 = addmod(v2, mulmod(contribution(kr, Hop{4, 2, 3}, h7),
   347	                                    mulmod(RHO, RHO)));
   348	    uint64_t v3t = v3 ^ 1;
   349	    std::snprintf(line, sizeof(line), "TRACE %s 1 1 3 0 1111 1 %llu", MAC,
   350	                  (unsigned long long)v1);
   351	    ctl.feed(line);
   352	    std::snprintf(line, sizeof(line), "TRACE %s 5 1 2 0 1111 2 %llu", MAC,
   353	                  (unsigned long long)v2);
   354	    ctl.feed(line);
   355	    std::snprintf(line, sizeof(line), "TRACE %s 4 2 3 0 1111 3 %llu", MAC,
   356	                  (unsigned long long)v3t);
   357	    ctl.feed(line);
   358	    std::snprintf(line, sizeof(line), "4 0 %llu %llu %s", n7,
   359	                  (unsigned long long)v3t, MAC);
   360	    Event ev7 = ctl.feed(line);
   361	    CHECK(!ev7.ok, "多故障包被拒");
   362	    CHECK_EQ(ev7.suspects.size(), 2, "多故障定位报出两个嫌疑");
   363	    CHECK_EQ(ev7.suspects.empty() ? -1 : ev7.suspects[0], 1, "改道嫌疑 s1");
   364	    CHECK_EQ(ev7.suspects.size() > 1 ? ev7.suspects[1] : -1, 4, "篡改嫌疑 s4");
   365	    CHECK_EQ(ev7.suspect_sid, 1, "首个嫌疑 = s1");
   366	  }
   367	
   368	  // 流级：F L O W 行一次判定
   369	  FlowAgg agg;
   370	  for (uint64_t i = 0; i < 10; ++i) {
   371	    Header x = forward(kr, p0, 0, 2000 + i);
   372	    agg.add(x.nonce, x.acc);
   373	    char pl[160];
   374	    std::snprintf(pl, sizeof(pl), "1 0 %llu %llu %s",
   375	                  (unsigned long long)x.nonce, (unsigned long long)x.acc, MAC);
   376	    ctl.feed(pl);
   377	  }
   378	  std::snprintf(line, sizeof(line), "FLOW %s 0 %llu %llu %llu", MAC,
   379	                (unsigned long long)agg.n, (unsigned long long)agg.sum_h,
   380	                (unsigned long long)agg.sum_v);
   381	  ev = ctl.feed(line);
   382	  CHECK(ev.kind == "flow" && ev.ok, "FLOW 行流级判定通过");
   383	  const DirStat& st = ctl.stat(0);
   384	  CHECK_EQ(st.cfmv_checks, 1, "流级判定仅 1 次");
   385	}
